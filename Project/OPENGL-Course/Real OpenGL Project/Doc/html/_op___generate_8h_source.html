72 MeshData GenerateCylinder(int segments, float size)
73 {
74 MeshData data;
75 float radius = size * 0.5f;
76 float height = size;
77 float halfH = height * 0.5f;
78
79 // Generate vertices for top and bottom circles
80 for (int i = 0; i <= segments; i++)
81 {
82 float angle = (float)i / (float)segments * 2.0f * (float)M_PI;

84 float z = sin(angle) * radius;

91 }
92
93 // Side triangles
94 for (int i = 0; i < segments; i++)
95 {

120 MeshData GenerateCone(int segments, float size)
121 {
122 MeshData data;
123 float radius = size * 0.5f;
124 float height = size;
125
126 // Apex
127 int apex = 0;
128 data.AddVertex(0, height, 0, 0.5f, 1.0f, 0, 1, 0, 0, 0, 0, 0, 0, 0);
129
130 // Base circle
131 for (int i = 0; i <= segments; i++)
132 {
133 float angle = (float)i / (float)segments * 2.0f * (float)M_PI;
134 float x = cos(angle) * radius;
135 float z = sin(angle) * radius;
136 float u = (float)i / (float)segments;
137
138 glm::vec3 sideNormal = glm::normalize(glm::vec3(x, radius, z));
139 data.AddVertex(x, 0, z, u, 0.0f, sideNormal.x, sideNormal.y, sideNormal.z, 0, 0, 0, 0, 0, 0);
140 }
141
142 // Side triangles
143 for (int i = 0; i < segments; i++)
144 {
145 data.AddTriangle(apex, i + 1, i + 2);
146 }

159 MeshData GenerateTorus(int segments, float size)
160 {
161 MeshData data;
162 float majorRadius = size * 0.5f;
163 float minorRadius = size * 0.15f;
164 int ringSegments = segments;
165 int tubeSegments = segments;
166
167 for (int i = 0; i <= ringSegments; i++)
168 {
169 float theta = (float)i / (float)ringSegments * 2.0f * (float)M_PI;
170 float cosTheta = cos(theta);
171 float sinTheta = sin(theta);
172
173 for (int j = 0; j <= tubeSegments; j++)
174 {
175 float phi = (float)j / (float)tubeSegments * 2.0f * (float)M_PI;
176 float cosPhi = cos(phi);
177 float sinPhi = sin(phi);
178
179 float x = (majorRadius + minorRadius * cosPhi) * cosTheta;
180 float y = minorRadius * sinPhi;
181 float z = (majorRadius + minorRadius * cosPhi) * sinTheta;
182
183 float nx = cosPhi * cosTheta;
184 float ny = sinPhi;
185 float nz = cosPhi * sinTheta;
186
187 float u = (float)i / (float)ringSegments;
188 float v = (float)j / (float)tubeSegments;
189
190 data.AddVertex(x, y, z, u, v, nx, ny, nz, 0, 0, 0, 0, 0, 0);
191 }
192 }
193
194 for (int i = 0; i < ringSegments; i++)
195 {
196 for (int j = 0; j < tubeSegments; j++)
197 {
198 int a = i * (tubeSegments + 1) + j;
199 int b = (i + 1) * (tubeSegments + 1) + j;
200 int c = (i + 1) * (tubeSegments + 1) + j + 1;
201 int d = i * (tubeSegments + 1) + j + 1;
202 data.AddTriangle(a, b, d);
203 data.AddTriangle(b, c, d);
204 }
205 }
206
207 return data;
208 }

240

243
244 for (int z = 0; z < resZ; z++)

246 for (int x = 0; x < resX; x++)
247 {
248 float px = x * spacing - halfW;
249 float pz = z * spacing - halfH;
250 float u = (float)x / (float)(resX - 1);
251 float v = (float)z / (float)(resZ - 1);
252
253 data.AddVertex(px, 0, pz, u, v, 0, 1, 0, 1, 0, 0, 0, 0, 1);
254 }
255 }
256
257 for (int z = 0; z < resZ - 1; z++)
258 {
259 for (int x = 0; x < resX - 1; x++)
260 {
261 int topLeft = z * resX + x;
262 int topRight = topLeft + 1;
263 int botLeft = (z + 1) * resX + x;
264 int botRight = botLeft + 1;
265
266 data.AddTriangle(topLeft, botLeft, topRight);
267 data.AddTriangle(topRight, botLeft, botRight);
