16class MeshOp_Transform : public Operation
17{
18public:
19 std::string GetName() const override { return "Transform"; }
20 std::string GetCategory() const override { return "Transform"; }
21
22 std::vector<ParamDef> GetParamDefs() const override
23 {
24 return {
25 ParamDef::Vec3("Translate", glm::vec3(0.0f)),
26 ParamDef::Vec3("Rotate", glm::vec3(0.0f)),
27 ParamDef::Vec3("Scale", glm::vec3(1.0f))
28 };
29 }
30
31 void Execute(OperationContext& ctx) override
32 {
33 if (ctx.mesh.vertices.empty()) return;
34
35 glm::vec3 translate = GetVec3("Translate");
36 glm::vec3 rotate = GetVec3("Rotate");
37 glm::vec3 scale = GetVec3("Scale");
38
39 glm::mat4 mat = glm::mat4(1.0f);
40 mat = glm::translate(mat, translate);
41 mat = glm::rotate(mat, glm::radians(rotate.x), glm::vec3(1, 0, 0));
42 mat = glm::rotate(mat, glm::radians(rotate.y), glm::vec3(0, 1, 0));
43 mat = glm::rotate(mat, glm::radians(rotate.z), glm::vec3(0, 0, 1));
44 mat = glm::scale(mat, scale);
45
46 ctx.mesh.TransformBy(mat);
47 }

57class MeshOp_Bend : public Operation
58{
59public:
60 std::string GetName() const override { return "Bend"; }
61 std::string GetCategory() const override { return "Transform"; }
62
63 std::vector<ParamDef> GetParamDefs() const override
64 {
65 return {
66 ParamDef::Enum("Axis", {"X", "Y", "Z"}, 1),
67 ParamDef::Float("Angle", 45.0f, -180.0f, 180.0f),
68 ParamDef::Float("Center", 0.0f, -10.0f, 10.0f)
69 };
70 }
71
72 void Execute(OperationContext& ctx) override
73 {
74 if (ctx.mesh.vertices.empty()) return;
75
76 int axis = GetEnum("Axis");
77 float angle = glm::radians(GetFloat("Angle"));
78 float center = GetFloat("Center");
79
80 // Get bounds along the bend axis to normalize the effect
81 glm::vec3 minB, maxB;
82 ctx.mesh.GetBounds(minB, maxB);
83
84 float axisMin, axisMax;
85 if (axis == 0) { axisMin = minB.x; axisMax = maxB.x; }
86 else if (axis == 1) { axisMin = minB.y; axisMax = maxB.y; }
87 else { axisMin = minB.z; axisMax = maxB.z; }
88
89 float range = axisMax - axisMin;
90 if (range < 0.0001f) return;
91
92 int count = ctx.mesh.GetVertexCount();
93 for (int i = 0; i < count; i++)
94 {
95 int base = i * 14;
96
97 float px = ctx.mesh.vertices[base];
98 float py = ctx.mesh.vertices[base + 1];
99 float pz = ctx.mesh.vertices[base + 2];
100
101 // Normalized parameter along the axis [0,1]
102 float t;
103 if (axis == 0) t = (px - axisMin) / range;
104 else if (axis == 1) t = (py - axisMin) / range;
105 else t = (pz - axisMin) / range;
106
107 float bendAngle = angle * (t - 0.5f); // Center the bend
108 float cosA = cos(bendAngle);
109 float sinA = sin(bendAngle);
110
111 // Bend: rotate the perpendicular axis around the bend center
112 if (axis == 1) // Bend Y-axis (most common — terrain bending)
113 {
114 float dist = pz - center;
115 ctx.mesh.vertices[base + 1] = py * cosA - dist * sinA;
116 ctx.mesh.vertices[base + 2] = py * sinA + dist * cosA + center;
117 }
118 else if (axis == 0)
119 {
120 float dist = pz - center;
121 ctx.mesh.vertices[base] = px * cosA - dist * sinA;
122 ctx.mesh.vertices[base + 2] = px * sinA + dist * cosA + center;
123 }
124 else
125 {
126 float dist = py - center;
127 ctx.mesh.vertices[base + 2] = pz * cosA - dist * sinA;
128 ctx.mesh.vertices[base + 1] = pz * sinA + dist * cosA + center;
129 }
130 }
131 }
132};
