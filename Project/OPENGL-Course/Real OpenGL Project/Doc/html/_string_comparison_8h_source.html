79inline unsigned int ASSIMP_itoa10(char *out, unsigned int max, int32_t number) {
80 ai_assert(nullptr != out);
81
82 // write the unary minus to indicate we have a negative number
83 unsigned int written = 1u;
84 if (number < 0 && written < max) {
85 *out++ = '-';
86 ++written;
87 number = -number;
88 }
89
90 // We begin with the largest number that is not zero.
91 int32_t cur = 1000000000; // 2147483648
92 bool mustPrint = false;
93 while (written < max) {
94
95 const unsigned int digit = number / cur;
96 if (mustPrint || digit > 0 || 1 == cur) {
97 // print all future zero's from now
98 mustPrint = true;
99
100 *out++ = '0' + static_cast<char>(digit);
101
102 ++written;
103 number -= digit * cur;
104 if (1 == cur) {
105 break;
106 }
107 }
108 cur /= 10;
109 }
110
111 // append a terminal zero
112 *out++ = '\0';
113 return written - 1;
114}

137inline int ASSIMP_stricmp(const char *s1, const char *s2) {
138 ai_assert(nullptr != s1);
139 ai_assert(nullptr != s2);
140
141#if (defined _MSC_VER)
142
143 return ::_stricmp(s1, s2);
144#else
145 char c1, c2;
146 do {
147 c1 = tolower((unsigned char)*(s1++));
148 c2 = tolower((unsigned char)*(s2++));
149 } while (c1 && (c1 == c2));
150 return c1 - c2;
151#endif
152}

163 return (i ? i : ASSIMP_stricmp(a.c_str(), b.c_str()));

178inline int ASSIMP_strincmp(const char *s1, const char *s2, unsigned int n) {
179 ai_assert(nullptr != s1);
180 ai_assert(nullptr != s2);
181 if (!n) {
182 return 0;
183 }
184
185#if (defined _MSC_VER)
186
187 return ::_strnicmp(s1, s2, n);
188
189#elif defined(__GNUC__)
190
191 return ::strncasecmp(s1, s2, n);
192
193#else
194 char c1, c2;
195 unsigned int p = 0;
196 do {
197 if (p++ >= n) return 0;
198 c1 = tolower((unsigned char)*(s1++));
199 c2 = tolower((unsigned char)*(s2++));
200 } while (c1 && (c1 == c2));
201
202 return c1 - c2;
203#endif
204}

212 unsigned int res = 1;

214 res *= base;
215 }
