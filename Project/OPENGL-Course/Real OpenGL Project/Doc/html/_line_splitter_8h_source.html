149AI_FORCE_INLINE LineSplitter::LineSplitter(StreamReaderLE& stream, bool skip_empty_lines, bool trim ) :
150 mIdx(0),
151 mCur(),
152 mEnd(nullptr),
153 mStream(stream),
154 mSwallow(),
155 mSkip_empty_lines(skip_empty_lines),
156 mTrim(trim) {
157 mCur.reserve(1024);
158 mEnd = mCur.c_str() + 1024;
159 operator++();
160 mIdx = 0;
161}

163AI_FORCE_INLINE LineSplitter& LineSplitter::operator++() {
164 if (mSwallow) {
165 mSwallow = false;
166 return *this;
167 }
168
169 if (!*this) {
170 throw std::logic_error("End of file, no more lines to be retrieved.");
171 }
172
173 char s;
174 mCur.clear();
175 while (mStream.GetRemainingSize() && (s = mStream.GetI1(), 1)) {
176 if (s == '\n' || s == '\r') {
177 if (mSkip_empty_lines) {
178 while (mStream.GetRemainingSize() && ((s = mStream.GetI1()) == ' ' || s == '\r' || s == '\n'));
179 if (mStream.GetRemainingSize()) {
180 mStream.IncPtr(-1);
181 }
182 } else {
183 // skip both potential line terminators but don't read past this line.
184 if (mStream.GetRemainingSize() && (s == '\r' && mStream.GetI1() != '\n')) {
185 mStream.IncPtr(-1);
186 }
187 if (mTrim) {
188 while (mStream.GetRemainingSize() && ((s = mStream.GetI1()) == ' ' || s == '\t'));
189 if (mStream.GetRemainingSize()) {
190 mStream.IncPtr(-1);
191 }
192 }
193 }
194 break;
195 }
196 mCur += s;
197 }
198 ++mIdx;
199
200 return *this;
201}

207AI_FORCE_INLINE const char *LineSplitter::operator[] (size_t idx) const {
208 const char* s = operator->()->c_str();
209
210 SkipSpaces(&s, mEnd);
211 for (size_t i = 0; i < idx; ++i) {
212 for (; !IsSpace(*s); ++s) {
213 if (IsLineEnd(*s)) {
214 throw std::range_error("Token index out of range, EOL reached");
215 }
216 }
217 SkipSpaces(&s, mEnd);
218 }
219 return s;
220}

222template <size_t N>
223AI_FORCE_INLINE void LineSplitter::get_tokens(const char* (&tokens)[N]) const {
224 const char* s = operator->()->c_str();
225
226 SkipSpaces(&s, mEnd);
227 for (size_t i = 0; i < N; ++i) {
228 if (IsLineEnd(*s)) {
229 throw std::range_error("Token count out of range, EOL reached");
230 }
231 tokens[i] = s;
232
233 for (; *s && !IsSpace(*s); ++s);
234 SkipSpaces(&s, mEnd);
235 }
236}

266AI_FORCE_INLINE bool LineSplitter::match_start(const char* check) {
267 const size_t len = ::strlen(check);
268
269 return len <= mCur.length() && std::equal(check, check + len, mCur.begin());
