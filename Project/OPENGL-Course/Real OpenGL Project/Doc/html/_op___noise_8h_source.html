22 static int perm[512];
23 static bool permInitialized = false;
24
25 inline void InitPerm(int seed)
26 {
27 srand(seed);
28 int base[256];
29 for (int i = 0; i < 256; i++) base[i] = i;
30 for (int i = 255; i > 0; i--)
31 {
32 int j = rand() % (i + 1);
33 int tmp = base[i]; base[i] = base[j]; base[j] = tmp;
34 }
35 for (int i = 0; i < 512; i++) perm[i] = base[i & 255];
36 permInitialized = true;
37 }

39 inline float Fade(float t) { return t * t * t * (t * (t * 6.0f - 15.0f) + 10.0f); }
40 inline float Lerp(float a, float b, float t) { return a + t * (b - a); }
41
42 inline float Grad(int hash, float x, float y)
43 {
44 int h = hash & 7;
45 float u = h < 4 ? x : y;
46 float v = h < 4 ? y : x;
47 return ((h & 1) ? -u : u) + ((h & 2) ? -v : v);
48 }
49
50 inline float Perlin2D(float x, float y)
51 {
52 int X = (int)floor(x) & 255;
53 int Y = (int)floor(y) & 255;
54 x -= floor(x); y -= floor(y);
55 float u = Fade(x), v = Fade(y);
56 int A = perm[X] + Y, B = perm[X + 1] + Y;
57 return Lerp(
58 Lerp(Grad(perm[A], x, y), Grad(perm[B], x - 1, y), u),
59 Lerp(Grad(perm[A + 1], x, y - 1), Grad(perm[B + 1], x - 1, y - 1), u),
60 v
61 );
62 }
63
64 inline float FractalNoise(float x, float y, int octaves, float persistence)
65 {
66 float total = 0; float amplitude = 1; float frequency = 1; float maxVal = 0;
67 for (int i = 0; i < octaves; i++)
68 {
69 total += Perlin2D(x * frequency, y * frequency) * amplitude;
70 maxVal += amplitude;
71 amplitude *= persistence;
72 frequency *= 2.0f;
73 }
74 return total / maxVal;
75 }
76}

137 int endV = (t == numThreads - 1) ? count : (t + 1) * perThread;
138

140 for (int i = startV; i < endV; i++)

143 float px = ctx.mesh.vertices[base];
144 float pz = ctx.mesh.vertices[base + 2];
145

149 octaves, persist

155 float ny = ctx.mesh.vertices[base + 6];

165 }
166 });

169 }

171 {
172 // Single-threaded path (small mesh or has vertex selection)
173 for (int i = 0; i < count; i++)
174 {
175 if (!ctx.IsVertexSelected(i)) continue;
176
177 int base = i * 14;
178 float px = ctx.mesh.vertices[base];
179 float pz = ctx.mesh.vertices[base + 2];
