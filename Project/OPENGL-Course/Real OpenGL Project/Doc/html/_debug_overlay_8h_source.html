12// Per-pass GPU timer (double-buffered async queries)
13struct PassTimer {
14 std::string name;
15 GLuint queries[2] = { 0, 0 };
16 int currentQuery = 0;
17 bool queryReady = false;
18 float timeMs = 0.0f;
19 float smoothedMs = 0.0f; // EMA smoothed
20 bool active = false; // Was this pass timed this frame?
21
22 void Init() {
23 if (!queries[0]) glGenQueries(2, queries);
24 }
25 void Begin() {
26 active = true;
27 // Read previous result
28 if (queryReady) {
29 GLuint64 gpuTime = 0;
30 GLint available = 0;
31 glGetQueryObjectiv(queries[1 - currentQuery], GL_QUERY_RESULT_AVAILABLE, &available);
32 if (available) {
33 glGetQueryObjectui64v(queries[1 - currentQuery], GL_QUERY_RESULT, &gpuTime);
34 timeMs = gpuTime / 1000000.0f;
35 smoothedMs = smoothedMs * 0.85f + timeMs * 0.15f; // EMA
36 }
37 }
38 glBeginQuery(GL_TIME_ELAPSED, queries[currentQuery]);
39 }
40 void End() {
41 glEndQuery(GL_TIME_ELAPSED);
42 currentQuery = 1 - currentQuery;
43 queryReady = true;
44 }
45 void Cleanup() {
46 if (queries[0]) { glDeleteQueries(2, queries); queries[0] = queries[1] = 0; }
47 }
48};
