43struct ParamValue

60 static ParamValue MakeEnum(int v) { ParamValue p; p.type = ParamType::Enum; p.enumVal = v; return p; }
61
62 // JSON Serialization
63 friend void to_json(json& j, const ParamValue& p) {
64 j = json{ {"type", p.type} };
65 switch (p.type) {
66 case ParamType::Float: j["value"] = p.floatVal; break;
67 case ParamType::Int: j["value"] = p.intVal; break;
68 case ParamType::Vec2: j["value"] = {p.vec2Val.x, p.vec2Val.y}; break;
69 case ParamType::Vec3: j["value"] = {p.vec3Val.x, p.vec3Val.y, p.vec3Val.z}; break;
70 case ParamType::Bool: j["value"] = p.boolVal; break;
71 case ParamType::Enum: j["value"] = p.enumVal; j["name"] = p.stringVal; break;
72 }
73 }
74
75 friend void from_json(const json& j, ParamValue& p) {
76 p.type = j.at("type").get<ParamType>();
77 if (j.contains("value")) {
78 switch (p.type) {
79 case ParamType::Float: p.floatVal = j.at("value").get<float>(); break;
80 case ParamType::Int: p.intVal = j.at("value").get<int>(); break;
81 case ParamType::Vec2: p.vec2Val = glm::vec2(j.at("value")[0], j.at("value")[1]); break;
82 case ParamType::Vec3: p.vec3Val = glm::vec3(j.at("value")[0], j.at("value")[1], j.at("value")[2]); break;
83 case ParamType::Bool: p.boolVal = j.at("value").get<bool>(); break;
84 case ParamType::Enum: p.enumVal = j.at("value").get<int>(); p.stringVal = j.value("name", ""); break;
85 }
86 }
