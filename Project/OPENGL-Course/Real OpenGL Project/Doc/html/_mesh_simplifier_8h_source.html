8// =====================================================================
9// MeshSimplifier — Automatic LOD Mesh Generation via Vertex Clustering
10//
11// Reduces polygon count by partitioning the mesh's bounding box into a
12// 3D spatial grid, merging all vertices within each cell into a single
13// representative vertex (averaged attributes), then rebuilding the
14// index buffer while discarding degenerate triangles.
15//
16// Usage:
17// MeshData lod1 = MeshSimplifier::Simplify(originalMesh, 0.5f); // ~50% reduction
18// MeshData lod2 = MeshSimplifier::Simplify(originalMesh, 0.25f); // ~75% reduction
19// =====================================================================
20class MeshSimplifier
21{
22public:
23 // Simplify a MeshData to approximately targetRatio of its original vertex count.
24 // targetRatio: 0.5 = keep ~50% of detail, 0.25 = keep ~25%, etc.
25 // Returns a new MeshData with reduced geometry. Returns empty MeshData on failure.
26 static MeshData Simplify(const MeshData& input, float targetRatio)
27 {
28 MeshData result;
29
30 int vertCount = input.GetVertexCount();
31 int triCount = input.GetTriangleCount();
32 if (vertCount < 4 || triCount < 2) return result; // Too few to simplify
33
34 targetRatio = std::max(0.05f, std::min(targetRatio, 1.0f));
35
36 // 1. Compute bounding box
37 glm::vec3 minB(1e10f), maxB(-1e10f);
38 for (int i = 0; i < vertCount; i++) {
39 glm::vec3 p = input.GetPosition(i);
40 minB = glm::min(minB, p);
41 maxB = glm::max(maxB, p);
42 }
43
44 glm::vec3 size = maxB - minB;
45
46 // Prevent degenerate dimensions (flat meshes like grass planes)
47 for (int i = 0; i < 3; i++) {
48 if (size[i] < 0.0001f) size[i] = 0.0001f;
49 }
50
51 // 2. Compute grid resolution
52 // We want roughly targetRatio * vertCount unique cells
53 // Grid is 3D, so resolution per axis = cbrt(targetVerts)
54 int targetVerts = std::max(4, (int)(vertCount * targetRatio));
55 int gridRes = std::max(2, (int)std::cbrt((double)targetVerts));
56
57 glm::vec3 cellSize = size / (float)gridRes;
58 for (int i = 0; i < 3; i++) {
59 if (cellSize[i] < 0.00001f) cellSize[i] = 0.00001f;
60 }
61
62 // 3. Accumulate vertex attributes per grid cell
63 static const int STRIDE = 14; // pos(3) + uv(2) + normal(3) + tangent(3) + bitangent(3)
64
65 struct CellAccum {
66 glm::vec3 posSum = glm::vec3(0.0f);
67 glm::vec2 uvSum = glm::vec2(0.0f);
68 glm::vec3 normalSum = glm::vec3(0.0f);
69 glm::vec3 tangentSum = glm::vec3(0.0f);
70 glm::vec3 bitangentSum = glm::vec3(0.0f);
71 int count = 0;
72 int newIndex = -1;
73 };
74
75 // Use int64 key for grid cell (allows large grids without collision)
76 std::map<int64_t, CellAccum> cells;
77 std::vector<int64_t> vertexCellKey(vertCount);
78
79 for (int v = 0; v < vertCount; v++) {
80 int base = v * STRIDE;
81 const float* d = &input.vertices[base];
82
83 glm::vec3 pos(d[0], d[1], d[2]);
84 glm::vec2 uv(d[3], d[4]);
85 glm::vec3 normal(d[5], d[6], d[7]);
86 glm::vec3 tangent(d[8], d[9], d[10]);
87 glm::vec3 bitangent(d[11], d[12], d[13]);
88
89 int cx = std::clamp((int)((pos.x - minB.x) / cellSize.x), 0, gridRes - 1);
90 int cy = std::clamp((int)((pos.y - minB.y) / cellSize.y), 0, gridRes - 1);
91 int cz = std::clamp((int)((pos.z - minB.z) / cellSize.z), 0, gridRes - 1);
92
93 int64_t key = (int64_t)cx * gridRes * gridRes + (int64_t)cy * gridRes + cz;
94 vertexCellKey[v] = key;
95
96 auto& cell = cells[key];
97 cell.posSum += pos;
98 cell.uvSum += uv;
99 cell.normalSum += normal;
100 cell.tangentSum += tangent;
101 cell.bitangentSum += bitangent;
102 cell.count++;
103 }
104
105 // 4. Build new vertex array from cell averages
106 int newVertIdx = 0;
107 for (auto& [key, cell] : cells) {
108 float inv = 1.0f / (float)cell.count;
109 glm::vec3 avgPos = cell.posSum * inv;
110 glm::vec2 avgUV = cell.uvSum * inv;
111
112 glm::vec3 avgNormal = glm::length(cell.normalSum) > 0.001f
113 ? glm::normalize(cell.normalSum) : glm::vec3(0.0f, 1.0f, 0.0f);
114 glm::vec3 avgTangent = glm::length(cell.tangentSum) > 0.001f
115 ? glm::normalize(cell.tangentSum) : glm::vec3(1.0f, 0.0f, 0.0f);
116 glm::vec3 avgBitangent = glm::length(cell.bitangentSum) > 0.001f
117 ? glm::normalize(cell.bitangentSum) : glm::vec3(0.0f, 0.0f, 1.0f);
118
119 result.AddVertex(
120 avgPos.x, avgPos.y, avgPos.z,
121 avgUV.x, avgUV.y,
122 avgNormal.x, avgNormal.y, avgNormal.z,
123 avgTangent.x, avgTangent.y, avgTangent.z,
124 avgBitangent.x, avgBitangent.y, avgBitangent.z
125 );
126
127 cell.newIndex = newVertIdx++;
128 }
129
130 // 5. Rebuild index buffer, discarding degenerate triangles
131 for (size_t i = 0; i + 2 < input.indices.size(); i += 3) {
132 unsigned int v0 = input.indices[i];
133 unsigned int v1 = input.indices[i + 1];
134 unsigned int v2 = input.indices[i + 2];
135
136 // Safety: skip out-of-range indices
137 if (v0 >= (unsigned int)vertCount || v1 >= (unsigned int)vertCount || v2 >= (unsigned int)vertCount)
138 continue;
139
140 int n0 = cells[vertexCellKey[v0]].newIndex;
141 int n1 = cells[vertexCellKey[v1]].newIndex;
142 int n2 = cells[vertexCellKey[v2]].newIndex;
143
144 // Skip degenerate triangles (all 3 verts collapsed into 1 or 2 cells)
145 if (n0 == n1 || n1 == n2 || n0 == n2) continue;
146
147 result.AddTriangle((unsigned int)n0, (unsigned int)n1, (unsigned int)n2);
148 }
149
150 // 6. Validate output — if simplification was too aggressive, return empty
151 if (result.GetVertexCount() < 3 || result.GetTriangleCount() < 1) {
152 result.Clear();
153 return result;
154 }
155
156 printf("[MeshSimplifier] %d verts / %d tris -> %d verts / %d tris (ratio=%.0f%%)\n",
157 vertCount, triCount,
158 result.GetVertexCount(), result.GetTriangleCount(),
159 targetRatio * 100.0f);
160
161 return result;
162 }
163};
