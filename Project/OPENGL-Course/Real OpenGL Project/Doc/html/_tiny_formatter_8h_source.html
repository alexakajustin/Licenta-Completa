70template < typename T,
71 typename CharTraits = std::char_traits<T>,
72 typename Allocator = std::allocator<T> >
73class basic_formatter {
74public:
75 typedef class std::basic_string<T,CharTraits,Allocator> string;
76 typedef class std::basic_ostringstream<T,CharTraits,Allocator> stringstream;
77
78 basic_formatter() {
79 // empty
80 }
81
82 /* Allow basic_formatter<T>'s to be used almost interchangeably
83 * with std::(w)string or const (w)char* arguments because the
84 * conversion c'tor is called implicitly. */
85 template <typename TT>
86 basic_formatter(const TT& sin) {
87 underlying << sin;
88 }
89
90 // Same problem as the copy constructor below, but with root cause is that stream move
91 // is not permitted on older GCC versions. Small performance impact on those platforms.
92#if defined(__GNUC__) && (__GNUC__ == 4 && __GNUC_MINOR__ <= 9)
93 basic_formatter(basic_formatter&& other) {
94 underlying << (string)other;
95 }
96#else
97 basic_formatter(basic_formatter&& other)
98 : underlying(std::move(other.underlying)) {
99 }
100#endif
101
102 // The problem described here:
103 // https://sourceforge.net/tracker/?func=detail&atid=1067632&aid=3358562&group_id=226462
104 // can also cause trouble here. Apparently, older gcc versions sometimes copy temporaries
105 // being bound to const ref& function parameters. Copying streams is not permitted, though.
106 // This workaround avoids this by manually specifying a copy ctor.
107#if !defined(__GNUC__) || !defined(__APPLE__) || __GNUC__ > 4 || (__GNUC__ == 4 && __GNUC_MINOR__ >= 6)
108 explicit basic_formatter(const basic_formatter& other) {
109 underlying << (string)other;
110 }
111#endif
112
113 operator string () const {
114 return underlying.str();
115 }
116
117 /* note - this is declared const because binding temporaries does only
118 * work for const references, so many function prototypes will
119 * include const basic_formatter<T>& s but might still want to
120 * modify the formatted string without the need for a full copy.*/
121 template <typename TToken, typename std::enable_if<!std::is_base_of<std::exception, TToken>::value>::type * = nullptr>
122 const basic_formatter &operator<<(const TToken &s) const {
123 underlying << s;
124 return *this;
125 }
126
127 template <typename TToken, typename std::enable_if<std::is_base_of<std::exception, TToken>::value>::type * = nullptr>
128 const basic_formatter &operator<<(const TToken &s) const {
129 underlying << s.what();
130 return *this;
131 }
132
133 template <typename TToken, typename std::enable_if<!std::is_base_of<std::exception, TToken>::value>::type * = nullptr>
134 basic_formatter &operator<<(const TToken &s) {
135 underlying << s;
136 return *this;
137 }
138
139 template <typename TToken, typename std::enable_if<std::is_base_of<std::exception, TToken>::value>::type * = nullptr>
140 basic_formatter &operator<<(const TToken &s) {
141 underlying << s.what();
142 return *this;
143 }
144
145
146 // comma operator overloaded as well, choose your preferred way.
147 template <typename TToken>
148 const basic_formatter& operator, (const TToken& s) const {
149 *this << s;
150 return *this;
151 }
152
153 template <typename TToken>
154 basic_formatter& operator, (const TToken& s) {
155 *this << s;
156 return *this;
157 }
158
159 // Fix for MSVC8
160 // See https://sourceforge.net/projects/assimp/forums/forum/817654/topic/4372824
161 template <typename TToken>
162 basic_formatter& operator, (TToken& s) {
163 *this << s;
164 return *this;
165 }
166
167
168private:
169 mutable stringstream underlying;
170};
