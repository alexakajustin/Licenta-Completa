169 // Helper to find the node associated with a bone in the scene
170 const aiNode *findBoneNode(const aiBone *bone) const {
171 if (bone == nullptr) {
172 return nullptr;
173 }
174
175 if (mName == bone->mName) {
176 return this;
177 }
178
179 for (unsigned int i = 0; i < mNumChildren; ++i) {
180 aiNode *aChild = mChildren[i];
181 if (aChild == nullptr) {
182 continue;
183 }
184
185 const aiNode *foundFromChild = nullptr;
186 foundFromChild = aChild->findBoneNode(bone);
187 if (foundFromChild) {
188 return foundFromChild;
189 }
190 }
191
192 return nullptr;
193 }

430 static const char* GetShortFilename(const char* filename) {
431 const char* lastSlash = strrchr(filename, '/');
432 const char* lastBackSlash = strrchr(filename, '\\');
433 if (lastSlash < lastBackSlash) {
434 lastSlash = lastBackSlash;
435 }
436 const char* shortFilename = lastSlash != nullptr ? lastSlash + 1 : filename;
437 return shortFilename;
438 }

446 std::pair<const aiTexture*, int> GetEmbeddedTextureAndIndex(const char* filename) const {
447 if (nullptr==filename) {
448 return std::make_pair(nullptr, -1);
449 }
450 // lookup using texture ID (if referenced like: "*1", "*2", etc.)
451 if ('*' == *filename) {
452 int index = std::atoi(filename + 1);
453 if (0 > index || mNumTextures <= static_cast<unsigned>(index)) {
454 return std::make_pair(nullptr, -1);
455 }
456 return std::make_pair(mTextures[index], index);
457 }
458 // lookup using filename
459 const char* shortFilename = GetShortFilename(filename);
460 if (nullptr == shortFilename) {
461 return std::make_pair(nullptr, -1);
462 }
463
464 for (unsigned int i = 0; i < mNumTextures; i++) {
465 const char* shortTextureFilename = GetShortFilename(mTextures[i]->mFilename.C_Str());
466 if (strcmp(shortTextureFilename, shortFilename) == 0) {
467 return std::make_pair(mTextures[i], static_cast<int>(i));
468 }
469 }
470 return std::make_pair(nullptr, -1);
471 }

479 inline aiBone *findBone(const aiString &name) const {
480 for (size_t m = 0; m < mNumMeshes; m++) {
481 aiMesh *mesh = mMeshes[m];
482 if (mesh == nullptr) {
483 continue;
484 }
485
486 for (size_t b = 0; b < mesh->mNumBones; b++) {
487 aiBone *bone = mesh->mBones[b];
488 if (bone == nullptr) {
489 continue;
490 }
491 if (name == bone->mName) {
492 return bone;
493 }
494 }
495 }
496 return nullptr;
497 }
