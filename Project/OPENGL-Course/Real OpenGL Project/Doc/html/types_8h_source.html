267struct aiString {
268#ifdef __cplusplus
270 aiString() AI_NO_EXCEPT :
271 length(0), data{'\0'} {
272#ifdef ASSIMP_BUILD_DEBUG
273 // Debug build: overwrite the string on its full length with ESC (27)
274 memset(data + 1, 27, AI_MAXLEN - 1);
275#endif
276 }
277
279 aiString(const aiString &rOther) :
280 length(rOther.length), data{'\0'} {
281 // Crop the string to the maximum length
282 length = length >= AI_MAXLEN ? AI_MAXLEN - 1 : length;
283 memcpy(data, rOther.data, length);
284 data[length] = '\0';
285 }
286
288 explicit aiString(const std::string &pString) :
289 length((ai_uint32)pString.length()), data{'\0'} {
290 length = length >= AI_MAXLEN ? AI_MAXLEN - 1 : length;
291 memcpy(data, pString.c_str(), length);
292 data[length] = '\0';
293 }
294
296 void Set(const std::string &pString) {
297 if (pString.length() > AI_MAXLEN - 1) {
298 return;
299 }
300 length = (ai_uint32)pString.length();
301 memcpy(data, pString.c_str(), length);
302 data[length] = 0;
303 }
304
306 void Set(const char *sz, size_t maxlen) {
307 if (sz == nullptr) {
308 return;
309 }
310 size_t len = 0;
311 for (size_t i=0; i<maxlen; ++i) {
312 if (sz[i] == '\0') {
313 break;
314 }
315 ++len;
316 }
317 if (len > AI_MAXLEN - 1) {
318 len = AI_MAXLEN - 1;
319 }
320 length = static_cast<uint32_t>(len);
321 memcpy(data, sz, len);
322 data[len] = 0;
323 }
324
326 aiString &operator=(const aiString &rOther) {
327 if (this == &rOther) {
328 return *this;
329 }
330
331 length = rOther.length;
332 if (length > (AI_MAXLEN - 1)) {
333 length = static_cast<ai_int32>(AI_MAXLEN - 1);
334 }
335
336 memcpy(data, rOther.data, length);
337 data[length] = '\0';
338 return *this;
339 }
340
342 aiString &operator=(const char *sz) {
343 Set(sz);
344 return *this;
345 }
346
348 aiString &operator=(const std::string &pString) {
349 Set(pString);
350 return *this;
351 }
352
354 bool operator==(const aiString &other) const {
355 if (length == other.length) {
356 return memcmp(data, other.data, length) == 0;
357 }
358 return false;
359 }
360
362 bool operator!=(const aiString &other) const {
363 return !(*this == other);
364 }
365
367 void Append(const char *app) {
368 const ai_uint32 len = static_cast<ai_uint32>(::strlen(app));
369 if (!len) {
370 return;
371 }
372 if (length + len >= AI_MAXLEN) {
373 return;
374 }
375
376 memcpy(&data[length], app, len + 1);
377 length += len;
378 }
379
381 void Clear() {
382 length = 0;
383 data[0] = '\0';
384
385#ifdef ASSIMP_BUILD_DEBUG
386 // Debug build: overwrite the string on its full length with ESC (27)
387 memset(data + 1, 27, AI_MAXLEN - 1);
388#endif
389 }
390
392 const char *C_Str() const {
393 return data;
394 }
395
400 bool Empty() const {
401 return length == 0;
402 }
403
404#endif // !__cplusplus
405
409 ai_uint32 length;
410
412 char data[AI_MAXLEN];
413}; // !struct aiString
