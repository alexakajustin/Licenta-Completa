243 float GetHeightAt(float x, float z, float radius = 0.0f) const
244 {
245 BuildHeightCacheIfNeeded();
246 if (!heightCache || heightCacheRes <= 0) return -1e10f;
247
248 float gx = (x - heightCacheMinX) / heightCacheCellSize;
249 float gz = (z - heightCacheMinZ) / heightCacheCellSize;
250
251 if (radius <= 0.0f) {
252 int ix = (int)gx, iz = (int)gz;
253 if (ix < 0 || ix >= heightCacheRes || iz < 0 || iz >= heightCacheResZ) return -1e10f;
254 return (*heightCache)[iz * heightCacheRes + ix];
255 }
256
257 // Radius search (find max height within radius)
258 int searchRange = (int)std::ceil(radius / heightCacheCellSize);
259 float radiusSq = radius * radius;
260 float maxH = -1e10f;
261 int ixCenter = (int)gx, izCenter = (int)gz;
262
263 for (int dz = -searchRange; dz <= searchRange; dz++) {
264 for (int dx = -searchRange; dx <= searchRange; dx++) {
265 int ix = ixCenter + dx, iz = izCenter + dz;
266 if (ix < 0 || ix >= heightCacheRes || iz < 0 || iz >= heightCacheResZ) continue;
267
268 float dxDist = (ix - gx) * heightCacheCellSize;
269 float dzDist = (iz - gz) * heightCacheCellSize;
270 if (dxDist * dxDist + dzDist * dzDist <= radiusSq) {
271 float h = (*heightCache)[iz * heightCacheRes + ix];
272 if (h > maxH) maxH = h;
273 }
274 }
275 }
276 return maxH;
277 }

320 void BuildHeightCacheIfNeeded() const
321 {
322 if (heightCacheBuilt.load(std::memory_order_acquire)) return;
323 std::lock_guard<std::mutex> lock(heightCacheMutex);
324 if (heightCacheBuilt.load(std::memory_order_relaxed)) return;
325 int vertCount = GetVertexCount();
326 int triCount = GetTriangleCount();
327 if (vertCount == 0 || triCount == 0) { heightCacheBuilt.store(true, std::memory_order_release); return; }
328
329 float minX = FLT_MAX, maxX = -FLT_MAX, minZ = FLT_MAX, maxZ = -FLT_MAX;
330 for (int i = 0; i < vertCount; i++) {
331 glm::vec3 p = GetPosition(i);
332 if (p.x < minX) minX = p.x; if (p.x > maxX) maxX = p.x;
333 if (p.z < minZ) minZ = p.z; if (p.z > maxZ) maxZ = p.z;
334 }
335
336 float rangeX = std::max(0.001f, maxX - minX), rangeZ = std::max(0.001f, maxZ - minZ);
337 float maxRange = std::max(rangeX, rangeZ);
338 int res = (int)glm::clamp(maxRange, 32.0f, 1024.0f);
339 if (maxRange < 50.0f && triCount > 1000) {
340 res = (triCount > 10000) ? 1024 : 512;
341 }
342 float cellSize = maxRange / (float)res;
343 int resX = std::min(res, (int)std::ceil(rangeX / cellSize));
344 int resZ = std::min(res, (int)std::ceil(rangeZ / cellSize));
345
346 auto* cache = new std::vector<float>(resX * resZ, -1e10f);
347 for (int t = 0; t < triCount; t++) {
348 glm::vec3 v0 = GetPosition(indices[t * 3]), v1 = GetPosition(indices[t * 3 + 1]), v2 = GetPosition(indices[t * 3 + 2]);
349 int gxMin = std::max(0, (int)((std::min({v0.x, v1.x, v2.x}) - minX) / cellSize));
350 int gxMax = std::min(resX - 1, (int)((std::max({v0.x, v1.x, v2.x}) - minX) / cellSize));
351 int gzMin = std::max(0, (int)((std::min({v0.z, v1.z, v2.z}) - minZ) / cellSize));
352 int gzMax = std::min(resZ - 1, (int)((std::max({v0.z, v1.z, v2.z}) - minZ) / cellSize));
353
354 glm::vec2 a2(v0.x, v0.z), b2(v1.x, v1.z), c2(v2.x, v2.z);
355 float denom = (b2.y - c2.y) * (a2.x - c2.x) + (c2.x - b2.x) * (a2.y - c2.y);
356 if (std::abs(denom) < 1e-10f) continue;
357 float invDenom = 1.0f / denom;
358
359 for (int gz = gzMin; gz <= gzMax; gz++) {
360 for (int gx = gxMin; gx <= gxMax; gx++) {
361 float px = minX + (gx + 0.5f) * cellSize, pz = minZ + (gz + 0.5f) * cellSize;
362 float u = ((b2.y - c2.y) * (px - c2.x) + (c2.x - b2.x) * (pz - c2.y)) * invDenom;
363 float v = ((c2.y - a2.y) * (px - c2.x) + (a2.x - c2.x) * (pz - c2.y)) * invDenom;
364 float w = 1.0f - u - v;
365 if (u >= -0.01f && v >= -0.01f && w >= -0.01f) {
366 float h = u * v0.y + v * v1.y + w * v2.y;
367 int idx = gz * resX + gx; if (h > (*cache)[idx]) (*cache)[idx] = h;
368 }
369 }
370 }
371 }
372
373 heightCacheMinX = minX; heightCacheMinZ = minZ; heightCacheCellSize = cellSize;
374 heightCacheRes = resX; heightCacheResZ = resZ; heightCache = cache;
375 heightCacheBuilt.store(true, std::memory_order_release);
376 }
