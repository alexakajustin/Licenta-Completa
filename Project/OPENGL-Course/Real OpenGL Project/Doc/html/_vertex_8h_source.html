66 // std::plus-family operates on operands with identical types - we need to
67 // support all the (vectype op float) combinations in vector maths.
68 // Providing T(float) would open the way to endless implicit conversions.
70 namespace Intern {
71 template <typename T0, typename T1, typename TRES = T0> struct plus {
72 TRES operator() (const T0& t0, const T1& t1) const {
73 return t0+t1;
74 }
75 };
76 template <typename T0, typename T1, typename TRES = T0> struct minus {
77 TRES operator() (const T0& t0, const T1& t1) const {
78 return t0-t1;
79 }
80 };
81 template <typename T0, typename T1, typename TRES = T0> struct multiplies {
82 TRES operator() (const T0& t0, const T1& t1) const {
83 return t0*t1;
84 }
85 };
86 template <typename T0, typename T1, typename TRES = T0> struct divides {
87 TRES operator() (const T0& t0, const T1& t1) const {
88 return t0/t1;
89 }
90 };
91 }

99struct Vertex {
100 friend Vertex operator + (const Vertex&,const Vertex&);
101 friend Vertex operator - (const Vertex&,const Vertex&);
102 friend Vertex operator * (const Vertex&,ai_real);
103 friend Vertex operator / (const Vertex&,ai_real);
104 friend Vertex operator * (ai_real, const Vertex&);
105
106 aiVector3D position;
107 aiVector3D normal;
108 aiVector3D tangent, bitangent;
109
110 aiVector3D texcoords[AI_MAX_NUMBER_OF_TEXTURECOORDS];
111 aiColor4D colors[AI_MAX_NUMBER_OF_COLOR_SETS];
112
113 Vertex() = default;
114 ~Vertex() = default;
115
116 // ----------------------------------------------------------------------------
118 explicit Vertex(const aiMesh* msh, unsigned int idx) {
119 ai_assert(idx < msh->mNumVertices);
120 position = msh->mVertices[idx];
121
122 if (msh->HasNormals()) {
123 normal = msh->mNormals[idx];
124 }
125
126 if (msh->HasTangentsAndBitangents()) {
127 tangent = msh->mTangents[idx];
128 bitangent = msh->mBitangents[idx];
129 }
130
131 for (unsigned int i = 0; msh->HasTextureCoords(i); ++i) {
132 texcoords[i] = msh->mTextureCoords[i][idx];
133 }
134
135 for (unsigned int i = 0; msh->HasVertexColors(i); ++i) {
136 colors[i] = msh->mColors[i][idx];
137 }
138 }
139

143 ai_assert(idx < msh->mNumVertices);

186 bool operator < (const Vertex & o) const {
187 if (position < o.position) return true;
188 if (position != o.position) return false;
189
190 if (normal < o.normal) return true;
191 if (normal != o.normal) return false;
192
193 for (uint32_t i = 0; i < AI_MAX_NUMBER_OF_TEXTURECOORDS; i ++) {
194 if (texcoords[i] < o.texcoords[i]) return true;
195 if (texcoords[i] != o.texcoords[i]) return false;
196 }
197
198 // note that tangent/bitangent are not checked since they are optional
199
200 for (uint32_t i = 0; i < AI_MAX_NUMBER_OF_COLOR_SETS; i ++) {
201 if (colors[i] < o.colors[i]) return true;
202 if (colors[i] != o.colors[i]) return false;
203 }
204
205 // if reached this point, they are equal
206 return false;
207 }
208
209 // ----------------------------------------------------------------------------
211 void SortBack(aiMesh* out, unsigned int idx) const {
212 ai_assert(idx<out->mNumVertices);
213 out->mVertices[idx] = position;
214
215 if (out->HasNormals()) {
216 out->mNormals[idx] = normal;
217 }
218
219 if (out->HasTangentsAndBitangents()) {
220 out->mTangents[idx] = tangent;
221 out->mBitangents[idx] = bitangent;
222 }
223
224 for(unsigned int i = 0; out->HasTextureCoords(i); ++i) {
225 out->mTextureCoords[i][idx] = texcoords[i];
226 }
227
228 for(unsigned int i = 0; out->HasVertexColors(i); ++i) {
229 out->mColors[i][idx] = colors[i];
230 }
231 }

237 template <template <typename t> class op> static Vertex BinaryOp(const Vertex& v0, const Vertex& v1) {
238 // this is a heavy task for the compiler to optimize ... *pray*
239
240 Vertex res;
241 res.position = op<aiVector3D>()(v0.position,v1.position);
242 res.normal = op<aiVector3D>()(v0.normal,v1.normal);
243 res.tangent = op<aiVector3D>()(v0.tangent,v1.tangent);
244 res.bitangent = op<aiVector3D>()(v0.bitangent,v1.bitangent);
245
246 for (unsigned int i = 0; i < AI_MAX_NUMBER_OF_TEXTURECOORDS; ++i) {
247 res.texcoords[i] = op<aiVector3D>()(v0.texcoords[i],v1.texcoords[i]);
248 }
249 for (unsigned int i = 0; i < AI_MAX_NUMBER_OF_COLOR_SETS; ++i) {
250 res.colors[i] = op<aiColor4D>()(v0.colors[i],v1.colors[i]);
251 }
252 return res;
253 }

257 template <template <typename, typename, typename> class op>
258 static Vertex BinaryOp(const Vertex& v0, ai_real f) {
259 // this is a heavy task for the compiler to optimize ... *pray*
260
261 Vertex res;
262 res.position = op<aiVector3D,ai_real,aiVector3D>()(v0.position,f);
263 res.normal = op<aiVector3D,ai_real,aiVector3D>()(v0.normal,f);
264 res.tangent = op<aiVector3D,ai_real,aiVector3D>()(v0.tangent,f);
265 res.bitangent = op<aiVector3D,ai_real,aiVector3D>()(v0.bitangent,f);
266
267 for (unsigned int i = 0; i < AI_MAX_NUMBER_OF_TEXTURECOORDS; ++i) {
268 res.texcoords[i] = op<aiVector3D,ai_real,aiVector3D>()(v0.texcoords[i],f);
269 }
270 for (unsigned int i = 0; i < AI_MAX_NUMBER_OF_COLOR_SETS; ++i) {
271 res.colors[i] = op<aiColor4D,float, aiColor4D>()(v0.colors[i],f);
272 }
273 return res;
274 }

278 template <template <typename, typename, typename> class op>
279 static Vertex BinaryOp(ai_real f, const Vertex& v0) {
280 // this is a heavy task for the compiler to optimize ... *pray*
281
282 Vertex res;
283 res.position = op<ai_real,aiVector3D,aiVector3D>()(f,v0.position);
284 res.normal = op<ai_real,aiVector3D,aiVector3D>()(f,v0.normal);
285 res.tangent = op<ai_real,aiVector3D,aiVector3D>()(f,v0.tangent);
286 res.bitangent = op<ai_real,aiVector3D,aiVector3D>()(f,v0.bitangent);
287
288 for (unsigned int i = 0; i < AI_MAX_NUMBER_OF_TEXTURECOORDS; ++i) {
289 res.texcoords[i] = op<ai_real,aiVector3D,aiVector3D>()(f,v0.texcoords[i]);
290 }
291 for (unsigned int i = 0; i < AI_MAX_NUMBER_OF_COLOR_SETS; ++i) {
292 res.colors[i] = op<float, aiColor4D,aiColor4D>()(f,v0.colors[i]);
293 }
294 return res;
