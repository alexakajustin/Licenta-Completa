59 // Extract frustum planes from a View-Projection matrix
60 // (Gribb/Hartmann method — standard, see OpenGL spec)
61 // Normals point INWARD (positive half-space = inside frustum)
62 // =================================================================
63 static Frustum CreateFrustumFromMatrix(const glm::mat4& viewProj) {
64 Frustum f;
65 // Transpose so we can work with column vectors as rows
66 glm::mat4 m = glm::transpose(viewProj);
67
68 // Near: row3 + row2
69 f.planes[0].normal = glm::vec3(m[3] + m[2]);
70 f.planes[0].d = m[3][3] + m[2][3];
71
72 // Far: row3 - row2
73 f.planes[1].normal = glm::vec3(m[3] - m[2]);
74 f.planes[1].d = m[3][3] - m[2][3];
75
76 // Left: row3 + row0
77 f.planes[2].normal = glm::vec3(m[3] + m[0]);
78 f.planes[2].d = m[3][3] + m[0][3];
79
80 // Right: row3 - row0
81 f.planes[3].normal = glm::vec3(m[3] - m[0]);
82 f.planes[3].d = m[3][3] - m[0][3];
83
84 // Bottom: row3 + row1
85 f.planes[4].normal = glm::vec3(m[3] + m[1]);
86 f.planes[4].d = m[3][3] + m[1][3];
87
88 // Top: row3 - row1
89 f.planes[5].normal = glm::vec3(m[3] - m[1]);
90 f.planes[5].d = m[3][3] - m[1][3];
91
92 // Normalize all planes
93 for (int i = 0; i < 6; i++) {
94 float len = glm::length(f.planes[i].normal);
95 if (len > 0.0f) {
96 f.planes[i].normal /= len;
97 f.planes[i].d /= len;
98 }
99 }

101 return f;
102 }

104 // =================================================================
105 // LAYER 1: Bounding Sphere vs Frustum
106 // Cheapest test — one dot product per plane.
107 // Returns false if the sphere is COMPLETELY outside any plane.
108 // =================================================================
109 bool IsSphereVisible(const glm::vec3& center, float radius) const {
110 for (int i = 0; i < 6; i++) {
111 // Add a small safety margin (2.0 units) to prevent popping at long distances
112 if (planes[i].SignedDistance(center) < -radius - 2.0f) {
113 return false; // Sphere is entirely behind this plane
114 }
115 }
116 return true;
117 }

119 // =================================================================
120 // LAYER 2: Contribution Culling (Screen-Space Size)
121 // Estimates the object's screen-space diameter in pixels.
122 // If it's smaller than minPixels, the object is too small to see.
123 //
124 // Formula: screenDiameter = (2 * radius * screenHeight) / (2 * dist * tan(fov/2))
125 // Since projection[1][1] = 1/tan(fov/2), we use that directly.
126 // =================================================================
127 static bool IsLargeEnough(const glm::vec3& sphereCenter, float sphereRadius,
128 float minPixels,
129 const glm::mat4& projection,
130 float screenHeight,
131 const glm::vec3& cameraPos) {
132 float dist = glm::length(sphereCenter - cameraPos);
133 if (dist < 0.001f) return true; // Camera is inside/on the object
134
135 // projection[1][1] = 1 / tan(fovY / 2)
136 float projFactor = projection[1][1];
137 float screenDiameter = (2.0f * sphereRadius * projFactor * screenHeight) / (2.0f * dist);
138
139 return screenDiameter >= minPixels;
140 }
141
142 // =================================================================
143 // LAYER 3: AABB vs Frustum (p-vertex method)
144 // Most precise test — tests the AABB corner most in the direction
145 // of each plane normal. If that corner is behind the plane, the
146 // entire box is outside.
147 // =================================================================
148 bool IsBoxVisible(const glm::vec3& min, const glm::vec3& max) const {
149 for (int i = 0; i < 6; i++) {
150 const Plane& p = planes[i];
151
152 // p-vertex: the corner of the AABB furthest along the plane normal
153 glm::vec3 pVertex = min;
154 if (p.normal.x >= 0) pVertex.x = max.x;
155 if (p.normal.y >= 0) pVertex.y = max.y;
156 if (p.normal.z >= 0) pVertex.z = max.z;
157
158 // Add a small safety margin (2.0 units) to prevent popping at long distances
159 if (p.SignedDistance(pVertex) < -2.0f) {
160 return false; // Box is entirely behind this plane
161 }
162 }
163 return true;
164 }

166 // =================================================================
167 // FULL PIPELINE: IsVisible()
168 // Chains all layers for maximum efficiency:
169 // 1. Sphere reject (cheap)
170 // 2. Contribution reject (trivial for visible objects)
171 // 3. AABB reject (precise, only if sphere didn't fully decide)
172 //
173 // Set minPixels <= 0 to disable contribution culling.
174 // Set screenHeight <= 0 to disable contribution culling.
175 // =================================================================
176 bool IsVisible(const glm::vec3& sphereCenter, float sphereRadius,
177 const glm::vec3& boxMin, const glm::vec3& boxMax,
178 float minPixels = 0.0f,
179 const glm::mat4& projection = glm::mat4(1.0f),
180 float screenHeight = 0.0f,
181 const glm::vec3& cameraPos = glm::vec3(0.0f)) const {
182
183 // Layer 1: Sphere test — cheapest, rejects most objects
184 if (!IsSphereVisible(sphereCenter, sphereRadius)) {
185 return false;
186 }
187
188 // Layer 2: Contribution culling — skip sub-pixel objects
189 if (minPixels > 0.0f && screenHeight > 0.0f) {
190 if (!IsLargeEnough(sphereCenter, sphereRadius, minPixels, projection, screenHeight, cameraPos)) {
191 return false;
192 }
193 }
194
195 // Layer 3: Precise AABB test — only for objects that passed sphere test
196 if (!IsBoxVisible(boxMin, boxMax)) {
197 return false;
198 }
199
200 return true;
201 }
